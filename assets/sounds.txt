# SFPS sound library -- edit freely, the dev build reloads this on save.
#
# No samples are stored. A sound is a handful of integers describing how to
# synthesise it, and the mixer builds the waveform as it plays. Same idea as
# the texture recipes: keep the instructions, not the result.
#
# Grammar (integers only, as everywhere else in this project):
#   s <name>                              begin a sound
#   l <wave> <ms> <f0> <f1> <atk> <dec> <vol>    add one layer
#
#   wave  0 square   1 saw   2 sine   3 noise
#   ms    how long this layer lasts
#   f0    starting frequency in Hz   (ignored by noise)
#   f1    ending frequency in Hz     -- the sweep is what gives a shot its
#                                       "pew" and an impact its "thud"
#   atk   attack in ms, from silence to full
#   dec   decay in ms, back to silence
#   vol   0-100
#
# Layers of one sound play together. Real weapon sounds are almost never a
# single oscillator: a shotgun is a noise burst for the crack plus a low
# square for the body, and leaving either one out sounds like a toy.

s shot
l 3 320 1400   60   0 300  90     # noise crack, falling fast
l 0 200  150   40   2 190  55     # low square body
l 1  90  420  110   0  90  35     # short saw bite up front

s impact
l 3 130  900  120   0 120  55     # short gritty tick
l 0  70  260   70   0  70  30     # small thump underneath

s pump
l 3  90 2200  700   0  85  40     # mechanical rasp
l 0  60  520  180   2  55  25

# --- monsters ------------------------------------------------------------
# Voiced with a saw for a growl and noise for breath. The imp is meant to
# read as an animal, so its calls sweep downward (a threat) rather than up.

s sight
l 1 260  180  260   6 240  70     # a rising snarl as it spots you
l 3 180  400  180  10 170  30     # breath under it

s eatt
l 1 150  320  120   0 140  75     # downward swipe growl
l 3 110  700  200   0 100  45     # claw-swish noise

s epain
l 1 120  520  200   0 110  70     # short yelp, falling
l 3  70  900  300   0  70  35

s edie
l 1 420  300   70   4 400  80     # long descending death groan
l 0 300  120   45  10 280  45     # low body
l 3 260  500  120   0 240  35     # rattle

s phurt
l 0  90  180  120   0  85  70     # a blunt grunt -- the player, not a monster
l 3  70  600  200   0  70  40

# --- pickups and the empty gun ------------------------------------------

s pammo
l 0 90  520  760   0  85  55     # a bright mechanical clack, rising
l 1 70  900 1300   0  65  30

s pmed
l 2 220 660  990   4 210  55     # a warm ascending chime
l 2 180 990 1320  10 170  40

# --- level exit ----------------------------------------------------------

s exit
l 1 320 300 1400   4 300  70    # a rising saw sweep -- you made it out
l 2 260 700 1600   8 240  45    # a sine over it for a chime
l 3 120 900  400   0 110  30    # a short zap

# --- game over, and game won -----------------------------------------------

# The player's death. Deliberately the inverse of `win` below: where that
# rises through three notes and resolves upward, this falls and never lands on
# anything. The two endings should be distinguishable with the screen off.
#
# Longer than every other sound here (900ms) because it is the only one nothing
# can interrupt -- the world is frozen behind it.
#
# 플레이어의 사망입니다. 아래의 `win`과 의도적으로 정반대입니다. 그쪽이 세 음을 거쳐
# 위로 해결되는 반면 이쪽은 하강하며 어디에도 안착하지 않습니다. 두 결말은 화면을 꺼도
# 구분되어야 합니다.
#
# 이곳의 다른 모든 소리보다 긴 900ms인데, 무엇도 이를 중단시킬 수 없는 유일한 소리이기
# 때문입니다. 뒤에서 월드가 정지해 있습니다.
s pdie
l 2 900 330  70    6 880  70     # a long sine fall, all the way down
l 0 700 160  40   10 680  50     # a square body sagging under it
l 3 420 500  90    0 400  40     # the breath going out

s win
l 2 260 440  660   4 240  60     # a rising sine fanfare, first note
l 2 300 660  990   8 280  65     # second note, a fifth up
l 2 460 990 1320  14 440  60     # third, resolving up an octave
l 0 700 220  110   0 650  20     # a long warm pad underneath

# --- the caster ------------------------------------------------------------
# Cold and tonal where the melee monsters are all growl and noise, so the sound
# of a bolt being readied is distinguishable from a claw swipe without looking.

s ecast
l 2 380 180  880   8 360  65     # a rising sine whine -- the wind-up you dodge
l 1 260 440 1320  20 240  40     # a saw over it for bite
l 3 120 300  900   0 110  25     # a breath of noise as it releases

s ehit
l 3 140 1200 260   0 130  60     # the bolt bursting
l 2 180  660 180   0 170  45     # a low tonal thump under it

# --- the grapple hook -------------------------------------------------------
# Four beats, four sounds, because the player has to be able to tell them apart
# without looking: the throw leaves, the claw bites something, the winch reels,
# and the arrival throws you off. An earlier version had only the throw and
# borrowed `impact`/`ehit` for the rest, which meant a hook landing on a wall
# was indistinguishable from a shotgun pellet doing the same.

s hook
l 0 55  2400  650   0  50  55     # a fast metallic snap -- the claw leaving
l 3 45  1600  300   0  40  35     # a scrape of noise riding it

# The claw biting STONE. Hard, bright, over instantly -- metal on rock. This is
# the sound that used to be the tail of `hook`, split out now that the flight
# has a real duration and the bite lands whenever it lands.
s hbite
l 0 70   900  220   0  65  55     # the clunk of the claw seating
l 3 60  2200  600   0  55  40     # a chip of stone coming off
l 0 110  150   90   4 100  35     # a low body under it, so it has weight

# The claw biting FLESH. Duller and lower with no bright chip: meat does not
# ring. Same length so the two read as the same event on different materials
# rather than as two different mechanics.
s hbiteb
l 3 90   420  110   0  85  60     # a wet thud
l 0 130  120   70   6 120  45     # a low body, longer than the stone version
l 2 70   260  140   0  65  25     # a soft tonal edge so it is not just noise

# The winch, reeling. The only LOOPING sound in the game: it is retriggered
# while HOOK_PULLING and must therefore be short enough to restart cleanly and
# quiet enough to sit under everything else. A rising sweep sells acceleration
# without needing to know the actual speed.
s hreel
l 1 110  180  340   6 100  32     # a saw rising -- the winch spinning up
l 3 90   600  900   0  85  18     # chain noise over it, deliberately faint

# Arrival: the launch that throws the player off. Bigger than the bite, with a
# downward sweep that reads as a shove rather than an impact -- this is the
# beat where the player stops being pulled and starts flying.
s hland
l 0 90   320   90   0  85  60     # a heavy thump
l 3 140 1400  200   0 130  45     # the burst of the release
l 2 220  180   60   8 210  40     # a low tail so the launch has follow-through
